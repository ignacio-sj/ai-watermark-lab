def merge_sorted_lists(list1, list2):
    """
    Merge two sorted lists into a single sorted list.

    Args:
        list1: A sorted list of comparable elements
        list2: A sorted list of comparable elements

    Returns:
        A single sorted list containing all elements from both lists

    >>> merge_sorted_lists([1, 3, 5], [2, 4, 6])
    [1, 2, 3, 4, 5, 6]

    >>> merge_sorted_lists([], [1, 2, 3])
    [1, 2, 3]

    >>> merge_sorted_lists([10], [5, 15, 20])
    [5, 10, 15, 20]
    """
    merged = []
    i = j = 0

    while i < len(list1) and j < len(list2):
        if list1[i] <= list2[j]:
            merged.append(list1[i])
            i += 1
        else:
            merged.append(list2[j])
            j += 1

    merged.extend(list1[i:])
    merged.extend(list2[j:])

    return merged
