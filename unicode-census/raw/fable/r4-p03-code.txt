def merge_sorted(left, right):
    """Merge two sorted lists into a single sorted list.

    Both inputs must already be sorted in ascending order. The inputs are
    not modified, and elements that compare equal keep their relative order
    (elements from ``left`` come first), so the merge is stable.

    >>> merge_sorted([1, 3, 5], [2, 4, 6])
    [1, 2, 3, 4, 5, 6]
    >>> merge_sorted([], [1, 2, 3])
    [1, 2, 3]
    >>> merge_sorted([1, 2, 2, 7], [2, 3])
    [1, 2, 2, 2, 3, 7]
    """
    merged = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
    merged.extend(left[i:])
    merged.extend(right[j:])
    return merged
