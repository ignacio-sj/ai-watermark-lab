def merge_sorted(left, right):
    """Merge two sorted lists into a single sorted list.

    Both input lists must already be sorted in ascending order. The
    result preserves the relative order of equal elements, taking from
    ``left`` first, so the merge is stable. The inputs are not modified.

    Args:
        left: A list sorted in ascending order.
        right: A list sorted in ascending order.

    Returns:
        A new list containing every element of ``left`` and ``right``
        in ascending order.

    >>> merge_sorted([1, 3, 5], [2, 4, 6])
    [1, 2, 3, 4, 5, 6]
    >>> merge_sorted([], [1, 2, 3])
    [1, 2, 3]
    >>> merge_sorted([1, 2, 2, 7], [2, 3])
    [1, 2, 2, 2, 3, 7]
    """
    merged = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
    merged.extend(left[i:])
    merged.extend(right[j:])
    return merged
