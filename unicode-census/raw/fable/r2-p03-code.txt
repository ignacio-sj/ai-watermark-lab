def merge_sorted(left, right):
    """Merge two sorted lists into a single sorted list.

    Both inputs must already be sorted in ascending order. A new list is
    returned and neither input is modified. Duplicates are preserved, and
    the merge is stable: when elements compare equal, those from ``left``
    appear before those from ``right``.

    >>> merge_sorted([1, 3, 5], [2, 4, 6])
    [1, 2, 3, 4, 5, 6]
    >>> merge_sorted([], [7, 8])
    [7, 8]
    >>> merge_sorted([1, 2, 2], [2, 3])
    [1, 2, 2, 2, 3]
    """
    merged = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
    merged.extend(left[i:])
    merged.extend(right[j:])
    return merged
